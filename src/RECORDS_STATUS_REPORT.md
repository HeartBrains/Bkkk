```typescript
// /utils/dateStatusHelper.ts
export function determineStatus(fromDate: string, toDate: string): Status {
  const today = new Date(2026, 2, 10); // March 10, 2026
  const from = new Date(fromDate);
  const to = toDate === 'Onwards' ? new Date(9999, 11, 31) : new Date(toDate);
  
  if (today < from) return 'upcoming';
  if (today >= from && today <= to) return 'current';
  return 'past';
}
```
